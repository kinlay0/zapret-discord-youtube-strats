; настройки ws-proxy. какой сервер использовать (cloudflare или deno) выбирается в service.bat - 8. WS Proxy
;
; port    - локальный порт прокси (только 127.0.0.1)
; setpac  - 1 = автоматически прописать PAC в настройки Windows (если вместе с этим используете локальный прокси, то поставьте 0)
; verbose - 1 = писать каждое соединение в окно прокси (ошибки пишутся всегда даже при 0)
;
; для сервера:
; url     - адрес сервера, обязательно вписываем с wss://
; key     - ваш ключ (значение переменной KEY (по простому, что писали в Value) в Worker'е/Deno)
; ip      - необязательно. подключаться к серверу по определенному IP вместо от DNS (узнать можно через Resolve-DnsName в PowerShell)
port=1080
setpac=1
verbose=0

[cloudflare]
; пример: wss://my-proxy.myname.workers.dev/
url=
key=
ip=

[deno]
; пример: wss://my-app.myname.deno.net/
url=
key=
ip=
